{{- /* The one sentence answer of an application page: what the buyer needs, in one quotable sentence, written from the specification tables. Stored on the page so the meta description, the visible answer block and the structured data all say the same thing. */ -}} {{- $page := .page -}} {{- partial "app-models.html" (dict "page" $page) -}} {{- $series := $page.Scratch.Get "appSeries" -}} {{- $models := $page.Scratch.Get "appModels" -}} {{- $loads := $page.Params.loads | default (slice) -}} {{- $factor := $page.Params.factor | default 1.1 -}} {{- $min := 0 }}{{ $max := 0 -}} {{- if $loads -}} {{- $min = index $loads 0 -}} {{- $max = index $loads (sub (len $loads) 1) -}} {{- end -}} {{- $lo := int (math.Ceil (math.Sub (math.Mul $min $factor) 0.0000001)) -}} {{- $hi := int (math.Ceil (math.Sub (math.Mul $max $factor) 0.0000001)) -}} {{- /* the series that cover the load range, the ones with the most models first */ -}} {{- $fit := slice -}} {{- range $series -}} {{- $st := .t -}} {{- $n := 0 -}} {{- range $models -}} {{- if and (eq .t $st) (ge .s $lo) (le .s $hi) -}} {{- $n = add $n 1 -}} {{- end -}} {{- end -}} {{- if gt $n 0 -}} {{- $t := .t -}} {{- $engine := partial "engine-brand.html" $t -}} {{- if not $engine -}} {{- if in (lower $t) " by " -}} {{- $words := split $t " " -}} {{- range $i, $w := $words -}} {{- if eq (lower $w) "by" -}} {{- if lt (add $i 1) (len $words) -}}{{- $engine = index $words (add $i 1) -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $fit = $fit | append (dict "engine" $engine "n" $n "t" .t "u" .u "hz" .hz "min" .min "max" .max) -}} {{- end -}} {{- end -}} {{- $fit = sort $fit "n" "desc" -}} {{- /* the engine names and the frequencies, the three biggest series */ -}} {{- $names := slice -}} {{- $hzList := slice -}} {{- $seen := "" -}} {{- range $fit -}} {{- $hzKey := printf "%v" .hz -}} {{- if not (strings.Contains $seen (printf "|%s|" $hzKey)) -}} {{- $seen = printf "%s|%s|" $seen $hzKey -}} {{- $hzList = $hzList | append $hzKey -}} {{- end -}} {{- end -}} {{- $seen = "" -}} {{- range $fit -}} {{- if and (lt (len $names) 3) (ne .engine "") (not (strings.Contains $seen (printf "|%s|" .engine))) -}} {{- $seen = printf "%s|%s|" $seen .engine -}} {{- $names = $names | append .engine -}} {{- end -}} {{- end -}} {{- $names = $names | first 3 -}} {{- $engines := delimit $names (i18n "list_sep") -}} {{- $hzText := "" -}} {{- if eq (len $hzList) 1 -}} {{- $hzText = printf "%v" (index $hzList 0) -}} {{- else if gt (len $hzList) 1 -}} {{- $hzText = printf (i18n "and_pair") (index $hzList 0) (index $hzList 1) -}} {{- end -}} {{- $name := $page.Params.app_name | default $page.Title -}} {{- $summary := printf (i18n "app_summary_tpl") $name $lo $hi $engines $hzText -}} {{- $page.Scratch.Set "appSummary" $summary -}} {{- $page.Scratch.Set "appRange" (dict "min" $lo "max" $hi "engines" $engines "hz" $hzText) -}} {{- $page.Scratch.Set "appFit" $fit -}}
