{{- /* The one sentence answer of an application page: what the buyer needs, in one quotable sentence, written from the specification tables. Stored on the page so the meta description, the visible answer block and the structured data all say the same thing. */ -}} {{- $page := .page -}} {{- partial "app-models.html" (dict "page" $page) -}} {{- $series := $page.Scratch.Get "appSeries" -}} {{- $models := $page.Scratch.Get "appModels" -}} {{- $loads := $page.Params.loads | default (slice) -}} {{- $factor := $page.Params.factor | default 1.1 -}} {{- $zh := eq site.Language.Lang "zh" -}} {{- $min := 0 }}{{ $max := 0 -}} {{- if $loads -}} {{- $min = index $loads 0 -}} {{- $max = index $loads (sub (len $loads) 1) -}} {{- end -}} {{- $lo := int (math.Ceil (math.Sub (math.Mul $min $factor) 0.0000001)) -}} {{- $hi := int (math.Ceil (math.Sub (math.Mul $max $factor) 0.0000001)) -}} {{- /* the series that cover the load range, the ones with the most models first */ -}} {{- $fit := slice -}} {{- range $series -}} {{- $st := .t -}} {{- $n := 0 -}} {{- range $models -}} {{- if and (eq .t $st) (ge .s $lo) (le .s $hi) -}} {{- $n = add $n 1 -}} {{- end -}} {{- end -}} {{- if gt $n 0 -}} {{- $engine := "" -}} {{- $t := .t -}} {{- if in (lower $t) " by " -}} {{- $words := split $t " " -}} {{- range $i, $w := $words -}} {{- if eq (lower $w) "by" -}} {{- if lt (add $i 1) (len $words) -}}{{- $engine = index $words (add $i 1) -}}{{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- $engine = replaceRE "^[0-9]+HZ\\s*" $t "" -}} {{- $engine = replaceRE "\\s*(发动机)?系列.*$" $engine "" -}} {{- end -}} {{- $fit = $fit | append (dict "engine" $engine "n" $n "t" .t "u" .u "hz" .hz "min" .min "max" .max) -}} {{- end -}} {{- end -}} {{- $fit = sort $fit "n" "desc" -}} {{- /* the engine names and the frequencies, the three biggest series */ -}} {{- $names := slice -}} {{- $hzList := slice -}} {{- $seen := "" -}} {{- range $fit -}} {{- $hzKey := printf "%v" .hz -}} {{- if not (strings.Contains $seen (printf "|%s|" $hzKey)) -}} {{- $seen = printf "%s|%s|" $seen $hzKey -}} {{- $hzList = $hzList | append $hzKey -}} {{- end -}} {{- end -}} {{- $seen = "" -}} {{- range $fit -}} {{- if and (lt (len $names) 3) (not (strings.Contains $seen (printf "|%s|" .engine))) -}} {{- $seen = printf "%s|%s|" $seen .engine -}} {{- $names = $names | append .engine -}} {{- end -}} {{- end -}} {{- $names = $names | first 3 -}} {{- $engines := delimit $names (cond $zh "、" ", ") -}} {{- $hzText := "" -}} {{- if eq (len $hzList) 1 -}} {{- $hzText = printf "%v" (index $hzList 0) -}} {{- else if gt (len $hzList) 1 -}} {{- $hzText = printf "%v and %v" (index $hzList 0) (index $hzList 1) -}} {{- end -}} {{- $summary := "" -}} {{- $name := $page.Params.app_name | default $page.Title -}} {{- if $zh -}} {{- $summary = printf "对于%[1]s，建议选用%[2]v到%[3]v kW的备用柴油发电机组；东方电机采用%[4]s发动机生产%[5]s Hz机型，可提供开式、静音罩壳、拖车、ATS与并联方案，出厂前每台机组均进行满载测试。" $name $lo $hi $engines $hzText -}} {{- else -}} {{- $summary = printf "For %[1]s the standby diesel generator set you need runs from %[2]v to %[3]v kW; Eastern Lion builds these sets around the %[4]s engines in %[5]s Hz, in open frame, silent canopy, trailer, ATS and parallel configurations, and every set is tested at full load before shipment." $name $lo $hi $engines $hzText -}} {{- end -}} {{- $page.Scratch.Set "appSummary" $summary -}} {{- $page.Scratch.Set "appRange" (dict "min" $lo "max" $hi "engines" $engines "hz" $hzText) -}} {{- $page.Scratch.Set "appFit" $fit -}}
