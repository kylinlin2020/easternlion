{{- /* the per series values of the decision guide and the one sentence answer, both derived from the specification table. Stored on the page so the guide, the summary and the search index share one source of truth. */ -}} {{- $page := .page -}} {{- $rows := $page.Params.rows | default (slice) -}} {{- $hz := $page.Params.hz | default 50 -}} {{- $kind := $page.Params.product_kind | default "genset" -}} {{- $min := $page.Params.power_min | default 0 -}} {{- $max := $page.Params.power_max | default 0 -}} {{- /* the engine brand from the series name, in any language */ -}} {{- $engine := partial "engine-brand.html" $page.Title -}} {{- if not $engine -}} {{- $words := split $page.Title " " -}} {{- $at := -1 -}} {{- range $i, $w := $words -}} {{- if eq (lower $w) "by" -}}{{- $at = $i -}}{{- end -}} {{- end -}} {{- if gt $at 0 -}}{{- $engine = index $words (add $at 1) | title -}}{{- end -}} {{- end -}} {{- /* the prime and the standby range out of the model rows */ -}} {{- $iP := -1 }}{{ $iS := -1 }} {{- range $i, $c := ($page.Params.columns | default (slice)) -}} {{- if eq $c "Prime power (kW)" -}}{{- $iP = $i -}}{{- end -}} {{- if eq $c "Standby power (kW)" -}}{{- $iS = $i -}}{{- end -}} {{- end -}} {{- $pMin := 999999.0 }}{{ $pMax := 0.0 }}{{ $sMin := 999999.0 }}{{ $sMax := 0.0 }} {{- if and (ge $iP 0) (ge $iS 0) -}} {{- range $rows -}} {{- $p := float (replaceRE "[^0-9.]" "" (printf "%v" (index . $iP))) -}} {{- $s := float (replaceRE "[^0-9.]" "" (printf "%v" (index . $iS))) -}} {{- if and (gt $p 0) (lt $p $pMin) -}}{{- $pMin = $p -}}{{- end -}} {{- if gt $p $pMax -}}{{- $pMax = $p -}}{{- end -}} {{- if and (gt $s 0) (lt $s $sMin) -}}{{- $sMin = $s -}}{{- end -}} {{- if gt $s $sMax -}}{{- $sMax = $s -}}{{- end -}} {{- end -}} {{- end -}} {{- if gt $pMin 999998.0 -}}{{- $pMin = 0.0 -}}{{- end -}} {{- if gt $sMin 999998.0 -}}{{- $sMin = 0.0 -}}{{- end -}} {{- $duty := "" -}} {{- if gt $sMax 0 -}} {{- $duty = printf (i18n "guide_duty_value") $sMin $sMax $pMin $pMax -}} {{- end -}} {{- $voltage := cond (eq $hz 60) (i18n "guide_voltage_60") (i18n "guide_voltage_50") -}} {{- $models := len $rows -}} {{- $engineValue := "" -}} {{- if $engine -}}{{- $engineValue = $engine -}}{{- else -}}{{- $engineValue = i18n "guide_engine_none" -}}{{- end -}} {{- if gt $models 0 -}} {{- $engineValue = printf "%s — %s" $engineValue (printf (i18n "guide_engine_count") $models) -}} {{- end -}} {{- /* the applications, for the one sentence answer */ -}} {{- $apps := "" -}} {{- $facts := index site.Data.series_facts (cond (eq $kind "alternator") "alternator" "genset") -}} {{- if $facts -}}{{- $apps = index $facts site.Language.Lang | default $facts.en -}}{{- end -}} {{- /* the one sentence answer — one template per shape, translated in i18n */ -}} {{- $kindLabel := cond (eq $kind "alternator") (i18n "kind_alternator") (i18n "kind_genset") -}} {{- $summary := "" -}} {{- if $engine -}} {{- $summary = printf (i18n "guide_sum_engine") $page.Title $hz $engine $min $max $models $apps -}} {{- else -}} {{- $summary = printf (i18n "guide_sum_plain") $page.Title $hz $kindLabel $min $max $models $apps -}} {{- end -}} {{- $page.Scratch.Set "guideSummary" $summary -}} {{- $page.Scratch.Set "guideValues" (dict "engine" $engineValue "duty" $duty "voltage" $voltage "models" $models "min" $min "max" $max "hz" $hz "kind" $kind "url" $page.RelPermalink "apps" $apps ) -}}
